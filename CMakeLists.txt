cmake_minimum_required(VERSION 3.22)
project(realm_cpp VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(CheckIPOSupported)
option(REALM_BUILD_VIEWER "Build the native graphics harness" ON)
option(REALM_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(REALM_FLASH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../realm-client" CACHE PATH "Authoritative Flash source")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(GLOB_RECURSE FLASH_INPUTS CONFIGURE_DEPENDS "${REALM_FLASH_ROOT}/src/*")
if(EXISTS "${REALM_FLASH_ROOT}/.git/logs/HEAD")
    list(APPEND FLASH_INPUTS "${REALM_FLASH_ROOT}/.git/logs/HEAD")
endif()
set(GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_custom_command(OUTPUT "${GENERATED_DIR}/flash_assets.hpp" "${GENERATED_DIR}/source_inventory.json"
    COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tools/import_flash.py"
        --flash-root "${REALM_FLASH_ROOT}" --output "${GENERATED_DIR}"
    DEPENDS ${FLASH_INPUTS} tools/import_flash.py VERBATIM)
add_custom_command(OUTPUT "${GENERATED_DIR}/flash_bitmap_colors.hpp"
    COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tools/compile_bitmap_colors.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/assets/bitmap-colors.bin" "${GENERATED_DIR}/flash_bitmap_colors.hpp"
    DEPENDS tools/compile_bitmap_colors.py assets/bitmap-colors.bin assets/bitmap-colors.json VERBATIM)
add_custom_command(OUTPUT "${GENERATED_DIR}/flash_party_transforms.hpp"
    COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tools/compile_party_transforms.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/assets/party-transforms.bin" "${GENERATED_DIR}/flash_party_transforms.hpp"
    DEPENDS tools/compile_party_transforms.py assets/party-transforms.bin assets/party-transforms.json VERBATIM)
add_custom_target(flash_manifest DEPENDS "${GENERATED_DIR}/flash_assets.hpp" "${GENERATED_DIR}/flash_bitmap_colors.hpp" "${GENERATED_DIR}/flash_party_transforms.hpp")

add_library(realm_core src/camera.cpp src/projectile.cpp src/protocol.cpp src/messages.cpp src/transport.cpp src/http.cpp)
target_include_directories(realm_core PUBLIC include "${GENERATED_DIR}")
add_dependencies(realm_core flash_manifest)
if(WIN32)
    target_link_libraries(realm_core PRIVATE ws2_32)
endif()
find_package(LibXml2 REQUIRED)
find_package(ZLIB REQUIRED)
find_package(nlohmann_json REQUIRED CONFIG)
add_library(realm_data src/catalog.cpp src/world.cpp src/session.cpp src/movement.cpp src/combat.cpp src/inventory.cpp src/party.cpp src/account.cpp)
target_link_libraries(realm_data PUBLIC realm_core PRIVATE LibXml2::LibXml2 ZLIB::ZLIB nlohmann_json::nlohmann_json)
if(MSVC)
    target_compile_options(realm_core PUBLIC /W4 /permissive-)
else()
    target_compile_options(realm_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(REALM_SANITIZERS AND NOT MSVC)
    target_compile_options(realm_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(realm_core PUBLIC -fsanitize=address,undefined)
endif()
check_ipo_supported(RESULT REALM_IPO_AVAILABLE OUTPUT REALM_IPO_ERROR)
if(REALM_IPO_AVAILABLE AND NOT REALM_SANITIZERS)
    set_property(TARGET realm_core PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()

add_executable(realm_session src/session_main.cpp)
target_link_libraries(realm_session PRIVATE realm_data)

find_package(PNG REQUIRED)
add_library(realm_assets src/image.cpp src/animation.cpp src/processed.cpp src/terrain.cpp src/font.cpp src/hud.cpp src/hud_art.cpp src/hud_input.cpp src/minimap.cpp src/tooltip.cpp src/player_tooltip.cpp src/interaction.cpp)
target_link_libraries(realm_assets PUBLIC realm_core realm_data PRIVATE PNG::PNG nlohmann_json::nlohmann_json)

add_executable(realm_terrain_inputs tools/terrain_inputs.cpp)
target_link_libraries(realm_terrain_inputs PRIVATE realm_assets nlohmann_json::nlohmann_json)

add_executable(realm_bake_inputs tools/bake_inputs.cpp)
target_link_libraries(realm_bake_inputs PRIVATE realm_assets realm_data nlohmann_json::nlohmann_json)
add_executable(realm_font_inputs tools/font_inputs.cpp)
target_link_libraries(realm_font_inputs PRIVATE realm_assets nlohmann_json::nlohmann_json)
add_executable(realm_hud_inputs tools/hud_inputs.cpp)
target_link_libraries(realm_hud_inputs PRIVATE realm_assets nlohmann_json::nlohmann_json)

add_executable(realm_minimap_inputs tools/minimap_inputs.cpp)
target_link_libraries(realm_minimap_inputs PRIVATE realm_assets nlohmann_json::nlohmann_json)
add_executable(realm_player_tooltip_inputs tools/player_tooltip_inputs.cpp)
target_link_libraries(realm_player_tooltip_inputs PRIVATE realm_assets nlohmann_json::nlohmann_json)

if(REALM_BUILD_VIEWER)
    find_package(SDL2 REQUIRED CONFIG)
    find_package(OpenGL REQUIRED)
    find_package(GLEW REQUIRED)
    add_executable(realm_viewer src/main.cpp src/renderer.cpp src/atlas.cpp src/scene.cpp src/startup.cpp)
    target_link_libraries(realm_viewer PRIVATE realm_data realm_assets SDL2::SDL2 OpenGL::GL GLEW::GLEW PNG::PNG)
    target_compile_definitions(realm_viewer PRIVATE SDL_MAIN_HANDLED)
    # The same program, starting with the Flash startup sequence instead of the gallery.
    add_executable(realm_client src/main.cpp src/renderer.cpp src/atlas.cpp src/scene.cpp src/startup.cpp)
    target_link_libraries(realm_client PRIVATE realm_data realm_assets SDL2::SDL2 OpenGL::GL GLEW::GLEW PNG::PNG)
    target_compile_definitions(realm_client PRIVATE SDL_MAIN_HANDLED REALM_CLIENT_DEFAULT=1)
    add_executable(realm_hud_gpu_inputs tools/hud_gpu_inputs.cpp src/renderer.cpp src/atlas.cpp)
    target_link_libraries(realm_hud_gpu_inputs PRIVATE realm_assets SDL2::SDL2 OpenGL::GL GLEW::GLEW nlohmann_json::nlohmann_json)
    if(REALM_IPO_AVAILABLE AND NOT REALM_SANITIZERS)
        set_property(TARGET realm_viewer realm_client PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
    endif()
endif()
